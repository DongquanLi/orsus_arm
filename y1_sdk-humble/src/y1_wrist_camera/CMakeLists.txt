cmake_minimum_required(VERSION 3.8)
project(y1_wrist_camera)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(wrist_camera_node src/wrist_camera_node.cpp)
target_compile_features(wrist_camera_node PUBLIC cxx_std_17)
target_compile_options(wrist_camera_node PRIVATE -Wall -Wextra -O2)
ament_target_dependencies(wrist_camera_node rclcpp sensor_msgs)
install(TARGETS wrist_camera_node DESTINATION lib/${PROJECT_NAME})
ament_package()
